CH 01 Abstract Data Types & STL
⭐ EXAM Abstract Data Type (ADT)

An Abstract Data Type (ADT) is a mathematical model for a data type defined entirely by its behavior from the user's perspective — not by its implementation.

  • Specifies possible values, possible operations, and behavior of those operations
  • Implementation details are hidden — this is the essence of abstraction
  • Focus on WHAT needs to be done, NOT HOW it is done
ADTOperations Defined
StackPUSH, POP, TOP, EMPTY, CREATE
QueueENQUEUE, DEQUEUE, FRONT, BACK, EMPTY
ListINSERT, DELETE, SEARCH, TRAVERSE
TreeINSERT, DELETE, SEARCH, TRAVERSALS
ADT vs Implementation:
ADT = "I want a stack that can push/pop" | Implementation = "here's the array/linked-list code"
Data Structure Operations
Core 4 Operations
  • Inserting — add new data
  • Deleting — remove data
  • Traversing — visit each element
  • Searching — find a specific element
Special Operations
  • Sorting — arrange in order
  • Merging — combine two structures

Vectors — C++ Code Segments

#include <vector>
#include <iostream>
using namespace std;

int main() {
    vector<int> v = {10, 20, 30, 40, 50};

    // ── INSERT at position (index 2) ──
    // Creates an iterator pointing to index 2
    auto it = v.begin() + 2;
    v.insert(it, 99);          // {10, 20, 99, 30, 40, 50}
    // All elements from index 2 onward shift RIGHT — O(n)

    // ── ERASE at position (index 2) ──
    it = v.begin() + 2;
    v.erase(it);               // {10, 20, 30, 40, 50}
    // All elements after erased position shift LEFT — O(n)

    // ── ERASE a range ──
    v.erase(v.begin() + 1, v.begin() + 3); // Remove index 1 and 2

    // ── PUSH / POP (end) ──
    v.push_back(60);           // O(amortized 1) — add to end
    v.pop_back();              // O(1) — remove from end

    // ── SIZE & CAPACITY (dynamic resizing) ──
    cout << v.size();          // Number of elements — O(1)
    cout << v.capacity();     // Allocated memory (doubles when full)

    // ── RESERVE (avoid reallocations) ──
    v.reserve(100);            // Pre-allocate 100 slots

    // ── TRAVERSE with iterator ──
    for (vector<int>::iterator i = v.begin(); i != v.end(); i++)
        cout << *i << " ";   // O(n)

    return 0;
}
Dynamic Resizing: When a vector's capacity is exceeded, it allocates a new block (~2× the old size), copies all elements there, and frees the old block. This makes individual push_back O(1) amortized, though occasional copies are O(n).

Rules for Simplifying Big-O
Rule 1: Drop constants
6n → O(n)
2N → O(N)
Rule 2: Drop low-order terms
O(N² + N + 1000) → O(N²)
Keep only the dominant term!
Special note: A loop running a fixed number of times (e.g., always 100 iterations) is still O(1) because it does NOT scale with input n.

class Node {
public:
    int no;
    Node *next;
};

// Create a node
Node *n = new Node();
n->no = 6;
n->next = NULL;

// List: head → [5|→] → [6|→] → [18|NULL]
void insertFront(int val) {
    Node *n = new Node();
    n->no = val;
    n->next = head;  // Step 1: new node → old head
    head = n;        // Step 2: update head
}
// Equivalent to push_front() in STL list — O(1)
void insertEnd(int val) {
    Node *n = new Node();
    n->no = val;
    n->next = NULL;

    if (head == NULL) { head = n; return; }

    Node *p = head;
    while (p->next != NULL) p = p->next; // traverse to last
    p->next = n;  // link last node to new node
}
// Equivalent to push_back() — O(n) without tail pointer
void insertMiddle(int val, int pos) {
    Node *n = new Node();
    n->no = val;
    n->next = NULL;

    Node *p = head;
    int i = 1;
    while (i < pos) { p = p->next; i++; } // stop 1 BEFORE insertion

    // ⚠️ CRITICAL ORDER — ALWAYS set n->next BEFORE p->next
    n->next = p->next;  // Step 1: new node forward link
    p->next = n;        // Step 2: previous node links to new
    // If you reverse this order, you LOSE the rest of the list!
}
// Mnemonic: "New-then-Prev"
⚠️ Critical order: Always set n->next = p->next BEFORE p->next = n, or you'll lose the rest of the list! 
// DELETE AT BEGINNING
void deleteFront() {
    if (head == NULL) return;
    Node *n = head;
    head = head->next;
    free(n);
}

// DELETE IN MIDDLE (by value)
void deleteMiddle(int target) {
    Node *n = head, *x;
    while (n->next->no != target) n = n->next; // stop 1 BEFORE target
    x = n->next;          // x = target node
    n->next = x->next;    // bypass target
    free(x);
}

// DELETE AT END
void deleteEnd() {
    Node *n = head, *x;
    while (n->next->next != NULL) n = n->next; // stop at 2nd-to-last
    x = n->next;
    n->next = NULL;
    free(x);
}

class Node {
public:
    int no;
    Node *next, *prev;
};

// ── INSERT AT BEGINNING ──
void insertFront(int val) {
    Node *n = new Node();
    n->no = val;
    n->next = head;
    n->prev = NULL;
    head->prev = n;   // old head points back to new node
    head = n;
}

// ── INSERT AT END ──
void insertEnd(int val) {
    Node *n = new Node();
    n->no = val;
    n->next = NULL;
    Node *p = head;
    while (p->next != NULL) p = p->next;
    p->next = n;   // forward link
    n->prev = p;   // backward link
}

// ── INSERT IN MIDDLE (after position pos) ──
void insertMiddle(int val, int pos) {
    Node *n = new Node();
    n->no = val;
    Node *p = head;
    int i = 1;
    while (i < pos) { p = p->next; i++; }

    n->next = p->next;       // forward links
    p->next->prev = n;       // node after p points back to n
    p->next = n;             // backward links
    n->prev = p;
}

// ── DELETE IN MIDDLE — Option 2 (cleaner, 1 pointer) ──
void deleteMiddle(int target) {
    Node *n = head;
    while (n->no != target) n = n->next; // go TO target directly
    n->prev->next = n->next;             // bypass forward
    n->next->prev = n->prev;             // bypass backward
    free(n);
}

// ── DELETE AT BEGINNING ──
void deleteFront() {
    Node *n = head;
    head = head->next;
    free(n);
    head->prev = NULL;  // fix backward pointer
}

⭐ EXAM Linked List Reversal — Iterative & Recursive

Iterative Reversal (Three-Pointer Method)

// Iterative — uses 3 pointers: prev, current, next
void reverseList(Node*& head) {
    Node *prev    = NULL;
    Node *current = head;
    Node *next    = NULL;

    while (current != NULL) {
        next          = current->next; // 1. Save next
        current->next = prev;          // 2. Reverse the link
        prev          = current;       // 3. Move prev forward
        current       = next;          // 4. Move current forward
    }
    head = prev; // New head is the old tail
}
// Mnemonic: "PNC" — Prev, Next, Current — 3 pointers to reverse
Step-by-step dry run for [1→2→3→NULL]:
1. prev=NULL, curr=1 → 1→NULL
2. prev=1, curr=2 → 2→1→NULL
3. prev=2, curr=3 → 3→2→1→NULL
4. curr=NULL → done. head=3 ✅

Recursive Reversal

// Recursive — reverses rest of list, then fixes link
Node* reverseRecursive(Node* head) {
    // Base case: empty or single node
    if (head == NULL || head->next == NULL) return head;

    // Recurse to end of list
    Node* newHead = reverseRecursive(head->next);

    // On unwind: make next node point BACK to current
    head->next->next = head;
    head->next       = NULL; // remove forward link

    return newHead; // return the new head (old tail)
}

// Usage: head = reverseRecursive(head);
Iterative: "PNC" — Prev, Next, Current — the three pointers needed
Recursive: Recurse to end → on unwind, flip each link

⭐ EXAM Application: Balanced Bracket Checker
bool isBalanced(string expr) {
    stack<char> s;
    for (char token : expr) {
        if (token == '(' || token == '{' || token == '[')
            s.push(token);  // Push all opening brackets
        else if (token == ')' || token == '}' || token == ']') {
            if (s.empty()) return false; // Nothing to match
            char top = s.top(); s.pop();
            // Verify the pair matches
            if ((token==')' && top!='(') ||
                (token=='}' && top!='{') ||
                (token==']' && top!='['))
                return false; // Mismatched pair
        }
    }
    return s.empty(); // Empty stack = all matched ✓
}
// Mnemonic: "POEM" — Push open, Open brackets, Empty=match, Match each close
🎮 Interactive Bracket Checker
Algorithm in plain English:
PUSH all opening brackets onto stack. On closing bracket, POP and CHECK it matches. After expression ends, stack must be EMPTY for balanced.
⭐ EXAM Application: Infix → Postfix Conversion

Infix: A + B (operators between operands — human readable)

Postfix: A B + (operators after operands — compiler friendly)

Postfix needs no parentheses and is simpler to evaluate (used by compilers).

Operator Priority

PriorityOperators
Higher* /
Lower+ -

Algorithm Summary

  • Operand (A,B,1,2…) → OUTPUT directly
  • '(' → PUSH onto stack
  • ')' → POP & OUTPUT until '(' found (discard '(')
  • Operator (+,−,*,/) → If stack empty OR higher priority than top: PUSH. Else: POP & OUTPUT, repeat comparison, then PUSH
  • End of expression → POP & OUTPUT all remaining items
Mnemonic: "DOPE" — Display operands, Open brackets get pushed, Pop on close, Empty stack at end

Postfix Evaluation Algorithm

// Evaluate postfix expression
int evalPostfix(string expr) {
    stack<int> s;
    for (char c : expr) {
        if (isdigit(c)) {
            s.push(c - '0');       // operand: push
        } else {                   // operator: pop 2, compute, push
            int b = s.top(); s.pop();
            int a = s.top(); s.pop();
            if (c == '+') s.push(a + b);
            else if (c == '-') s.push(a - b);
            else if (c == '*') s.push(a * b);
            else if (c == '/') s.push(a / b);
        }
    }
    return s.top(); // final result
}
// Dry run "2 4 * 9 5 + –" → 2×4=8, 9+5=14, 8–14=–6

// Enqueue (insert at TAIL)
void enqueue(int val) {
    Node *n = new Node();
    n->data = val; n->next = NULL;
    if (head == NULL) { head = tail = n; return; }
    tail->next = n;
    tail = n;           // update tail pointer
}

// Dequeue (remove from HEAD)
void dequeue() {
    if (head == NULL) return;
    Node *n = head;
    head = head->next;
    free(n);
    if (head == NULL) tail = NULL; // list is now empty
}

// Postorder (Left → Right → Root)
void postorder(Node *node) {
    if (node == NULL) return;
    postorder(node->left);   // Left
    postorder(node->right);  // Right
    cout << node->data << " "; // Root LAST
}

// Preorder (Root → Left → Right)
void preorder(Node *node) {
    if (node == NULL) return;
    cout << node->data << " "; // Root FIRST
    preorder(node->left);
    preorder(node->right);
}

// Inorder (Left → Root → Right)
void inorder(Node *node) {
    if (node == NULL) return;
    inorder(node->left);
    cout << node->data << " "; // Root MIDDLE
    inorder(node->right);
}

PDF Sorting Notes
Insertion Sort

Builds the sorted list one item at a time. Mnemonic from the notes: KICK — Key stored, Iterate backwards, Compare and shift, Key placed.

Selection Sort

Repeatedly finds the minimum in the unsorted portion and swaps it into the sorted boundary.

Bubble Sort

Compares adjacent pairs and swaps them until no swaps are needed.

Heap Sort

Uses a max-heap: build heap, exchange root with last unsorted item, then reheap down.

Quick Sort

Uses divide and conquer with the last element as pivot, then recursively sorts left and right partitions.

Merge Sort

Divide, sort left, sort right, then merge. Mnemonic from the notes: DSMC — Divide, Sort, Merge, Combined.
